/* =========================================================
   1. LEAP YEAR
   ========================================================= */
#include <stdio.h>
int main() {
    int y;
    scanf("%d", &y);
    if((y%4==0 && y%100!=0) || y%400==0)
        printf("Leap Year");
    else
        printf("Not Leap Year");
    return 0;
}

/* =========================================================
   4. SALARY CALCULATION
   ========================================================= */
#include <stdio.h>
int main(){
    float basic,hra,ta,gross,pt,net;
    scanf("%f",&basic);
    hra=0.1*basic;
    ta=0.05*basic;
    gross=basic+hra+ta;
    pt=0.02*gross;
    net=gross-pt;
    printf("Net Salary=%.2f",net);
    return 0;
}


/* =========================================================
   5. MATRIX ADDITION
   ========================================================= */
#include <stdio.h>
int main(){
    int a[2][2],b[2][2],c[2][2],i,j;
    for(i=0;i<2;i++)
        for(j=0;j<2;j++) scanf("%d",&a[i][j]);
    for(i=0;i<2;i++)
        for(j=0;j<2;j++) scanf("%d",&b[i][j]);

    for(i=0;i<2;i++){
        for(j=0;j<2;j++){
            c[i][j]=a[i][j]+b[i][j];
            printf("%d ",c[i][j]);
        }
        printf("\n");
    }
    return 0;
}


/* =========================================================
   6. FACTORIAL (RECURSION)
   ========================================================= */
#include <stdio.h>
int fact(int n){
    if(n==0) return 1;
    return n*fact(n-1);
}
int main(){
    int n;
    scanf("%d",&n);
    printf("%d",fact(n));
    return 0;
}

/* =========================================================
   8. STUDENT RESULT
   ========================================================= */
#include <stdio.h>
int main(){
    int m[5],i,sum=0;
    for(i=0;i<5;i++){ scanf("%d",&m[i]); sum+=m[i]; }
    float avg=sum/5.0;

    if(avg>=75) printf("Distinction");
    else if(avg>=60) printf("First");
    else if(avg>=50) printf("Second");
    else if(avg>=40) printf("Third");
    else printf("Fail");
    return 0;
}


/* =========================================================
   9. ARMSTRONG NUMBER
   ========================================================= */
#include <stdio.h>
int main(){
    int n,r,sum=0,temp;
    scanf("%d",&n);
    temp=n;
    while(n>0){
        r=n%10;
        sum+=r*r*r;
        n/=10;
    }
    if(sum==temp) printf("Armstrong");
    else printf("Not Armstrong");
    return 0;
}

/* =========================================================
   12. GCD
   ========================================================= */
#include <stdio.h>
int main(){
    int a,b,i,gcd;
    scanf("%d%d",&a,&b);
    for(i=1;i<=a&&i<=b;i++)
        if(a%i==0 && b%i==0) gcd=i;
    printf("%d",gcd);
    return 0;
}

/* =========================================================
   14. BINARY TO DECIMAL
   ========================================================= */
#include <stdio.h>
#include <math.h>
int main(){
    int n,i=0,rem,dec=0;
    scanf("%d",&n);
    while(n>0){
        rem=n%10;
        dec+=rem*pow(2,i);
        n/=10;
        i++;
    }
    printf("%d",dec);
    return 0;
}

/* =========================================================

    FILE *f1,*f2;

    f1=fopen("a.txt","r");
    f2=fopen("b.txt","w");

    fclose(f1);

    return 0;
}
